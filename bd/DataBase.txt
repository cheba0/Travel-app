-- Создание таблицы Пользователи
CREATE TABLE Пользователи (
    id SERIAL,
    email VARCHAR(255) NOT NULL UNIQUE,
    имя_пользователя VARCHAR(100) NOT NULL,
    хеш_пароля VARCHAR(255) NOT NULL,
    CONSTRAINT пользователи_id_pk PRIMARY KEY(id)
);

-- Создание таблицы Путешествия
CREATE TABLE Путешествия (
    id SERIAL,
    имя_путешествия VARCHAR(255) NOT NULL,
    описание TEXT,
    дата_начала DATE NOT NULL,
    дата_окончания DATE NOT NULL,
    создатель INT NOT NULL,
    CONSTRAINT путешествия_id_pk PRIMARY KEY(id),
    CONSTRAINT путешествия_создатель_fk FOREIGN KEY(создатель) REFERENCES Пользователи(id),
    CONSTRAINT chk_dates_valid CHECK (дата_окончания >= дата_начала)
);

-- Создание таблицы Участники_путешествий
CREATE TABLE Участники_путешествий (
    id SERIAL,
    путешествие INT NOT NULL,
    участник INT NOT NULL,
    активность BOOLEAN NOT NULL DEFAULT TRUE,
    CONSTRAINT участники_путешествий_id_pk PRIMARY KEY(id),
    CONSTRAINT участники_путешествие_fk FOREIGN KEY(путешествие) REFERENCES Путешествия(id),
    CONSTRAINT участники_участник_fk FOREIGN KEY(участник) REFERENCES Пользователи(id),
    CONSTRAINT unique_participant UNIQUE(путешествие, участник)
);

-- Создание таблицы Расходы
CREATE TABLE Расходы (
    id SERIAL,
    путешествие INT NOT NULL,
    кто_оплатил INT NOT NULL,
    категория VARCHAR(100) NOT NULL,
    сумма DECIMAL(10, 2) NOT NULL,
    описание TEXT,
    дата DATE NOT NULL,
    CONSTRAINT расходы_id_pk PRIMARY KEY(id),
    CONSTRAINT расходы_путешествие_fk FOREIGN KEY(путешествие) REFERENCES Путешествия(id),
    CONSTRAINT расходы_кто_оплатил_fk FOREIGN KEY(кто_оплатил) REFERENCES Пользователи(id),
    CONSTRAINT chk_positive_amount CHECK (сумма > 0)
);

-- Создание таблицы Распределение_расхода
CREATE TABLE Распределение_расхода (
    id SERIAL,
    расход INT NOT NULL,
    участник INT NOT NULL,
    сумма_доли DECIMAL(10, 2) NOT NULL,
    процент_доли DECIMAL(5, 2) NOT NULL,
    CONSTRAINT распределение_расхода_id_pk PRIMARY KEY(id),
    CONSTRAINT распределение_расход_fk FOREIGN KEY(расход) REFERENCES Расходы(id) ON DELETE CASCADE,
    CONSTRAINT распределение_участник_fk FOREIGN KEY(участник) REFERENCES Пользователи(id),
    CONSTRAINT chk_percentage_valid CHECK (процент_доли BETWEEN 0 AND 100),
    CONSTRAINT chk_share_amount_valid CHECK (сумма_доли >= 0)
);

-- Создание таблицы Долги
CREATE TABLE Долги (
    id SERIAL,
    путешествие INT NOT NULL,
    кредитор INT NOT NULL,
    должник INT NOT NULL,
    общая_сумма DECIMAL(10, 2) NOT NULL,
    статус_оплаты BOOLEAN NOT NULL DEFAULT FALSE,
    CONSTRAINT долги_id_pk PRIMARY KEY(id),
    CONSTRAINT долги_путешествие_fk FOREIGN KEY(путешествие) REFERENCES Путешествия(id),
    CONSTRAINT долги_кредитор_fk FOREIGN KEY(кредитор) REFERENCES Пользователи(id),
    CONSTRAINT долги_должник_fk FOREIGN KEY(должник) REFERENCES Пользователи(id),
    CONSTRAINT chk_positive_debt CHECK (общая_сумма > 0),
    CONSTRAINT chk_different_users CHECK (кредитор != должник)
);
